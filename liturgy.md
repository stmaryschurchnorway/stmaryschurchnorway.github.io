liturgies:
  - id: "holy-qurbana"
    title: "Holy Qurbana"

    malayalam: "വിശുദ്ധ കുർബാന"
    transliteration: "Vi-shud-dha Kur-ba-na"
    description: "The Eucharist—'sacrament of sacraments'—where bread and wine become the Body and Blood of Christ through the Holy Spirit's invocation."

    image: "/assets/img/qurbana-liturgy.png"
    url: "/qurbana/"

  - id: "holy-baptism"
    title: "Holy Baptism"

    malayalam: "മാമോദീസാ"
    transliteration: "Ma-mo-dee-sa"
    description: "The Gate to the Kingdom—washing away original sin and incorporating the believer into Christ's death and resurrection through water and the Holy Spirit."

    image: "/assets/img/mamodisa-liturgy.png"
    url: "/mamodisa/"

  - id: "holy-mooron"
    title: "Holy Mooron"

    malayalam: "മൂറോൻ"
    transliteration: "Moo-ron"
    description: "Chrismation—the seal of the Holy Spirit that completes baptism, anointing the believer as priest, prophet, and king in God's Kingdom."

    image: "/assets/img/mooron-liturgy.png"
    url: "/mooron/"

  - id: "holy-confession"
    title: "Holy Confession"

    malayalam: "കുമ്പസാരം"
    transliteration: "Kum-pa-sa-ram"
    description: "The sacrament of repentance and reconciliation — where we return to God, receive His forgiveness, and are restored. The Church Fathers called it a 'second baptism.'"

    image: "/assets/img/confession-liturgy.png"
    url: "/confession/"

  - id: "anointing-sick"
    title: "Anointing of the Sick"

    malayalam: "എണ്ണ തേച്ചു പ്രാർത്ഥിക്കൽ"
    transliteration: "En-na the-chu prar-thi-kal"
    description: "Holy Unction—prayers and anointing with blessed oil for healing and comfort of those suffering in body, mind, or spirit."

    image: "/assets/img/anointing-sick-liturgy.png"
    url: "/anointing/"

  - id: "holy-matrimony"
    title: "Holy Matrimony"

    malayalam: "വിവാഹം"
    transliteration: "Vi-va-ham"
    description: "Marriage—holy union blessed by the Church as an icon of Christ and the Church, sanctifying the couple's life together."

    image: "/assets/img/matrimony-liturgy.png"
    url: "/matrimony/"

  - id: "holy-orders"
    title: "Holy Orders"

    malayalam: "പൗരോഹിത്യം"
    transliteration: "Pou-ro-hi-thyam"
    description: "Priesthood—ordination to serve the Church in apostolic succession, continuing Christ's ministry of word and sacrament."

    image: "/assets/img/orders-liturgy.png"
    url: "/holy-orders/"
