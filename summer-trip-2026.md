<script>
(function() {
  // --- Countdown ---
  var target = new Date('2026-05-22T18:00:00+02:00').getTime();
  function tick() {
    var now = Date.now();
    var diff = target - now;
    if (diff <= 0) {
      document.getElementById('cd-days').textContent = '0';
      document.getElementById('cd-hours').textContent = '0';
      document.getElementById('cd-mins').textContent = '0';
      document.getElementById('cd-secs').textContent = '0';
      return;
    }
    var d = Math.floor(diff / 86400000);
    var h = Math.floor((diff % 86400000) / 3600000);
    var m = Math.floor((diff % 3600000) / 60000);
    var s = Math.floor((diff % 60000) / 1000);
    document.getElementById('cd-days').textContent = d;
    document.getElementById('cd-hours').textContent = h;
    document.getElementById('cd-mins').textContent = m;
    document.getElementById('cd-secs').textContent = s;
  }
  tick();
  setInterval(tick, 1000);

  // --- Day tabs ---
  var tabs = document.querySelectorAll('.st-day-tab');
  tabs.forEach(function(tab) {
    tab.addEventListener('click', function() {
      tabs.forEach(function(t) { t.classList.remove('active'); });
      tab.classList.add('active');
      document.querySelectorAll('.st-schedule-panel').forEach(function(p) { p.classList.remove('active'); });
      document.getElementById('panel-' + tab.getAttribute('data-day')).classList.add('active');
    });
  });

  // --- Scroll reveal ---
  var reveals = document.querySelectorAll('.st-reveal');
  var observer = new IntersectionObserver(function(entries) {
    entries.forEach(function(entry) {
      if (entry.isIntersecting) {
        entry.target.classList.add('visible');
        observer.unobserve(entry.target);
      }
    });
  }, { threshold: 0.08 });
  reveals.forEach(function(el) { observer.observe(el); });

  // --- Active nav highlight ---
  var navLinks = document.querySelectorAll('.st-nav a');
  var sections = [];
  navLinks.forEach(function(link) {
    var id = link.getAttribute('href').replace('#', '');
    var sec = document.getElementById(id);
    if (sec) sections.push({ el: sec, link: link });
  });
  window.addEventListener('scroll', function() {
    var scrollY = window.scrollY + 200;
    var current = null;
    sections.forEach(function(s) {
      if (s.el.offsetTop <= scrollY) current = s;
    });
    navLinks.forEach(function(l) { l.classList.remove('active'); });
    if (current) current.link.classList.add('active');
  });

  // --- Checklist ---
  document.querySelectorAll('.st-check').forEach(function(box) {
    function toggle() {
      var checked = box.classList.toggle('checked');
      box.setAttribute('aria-checked', checked);
      box.parentElement.classList.toggle('checked-item', checked);
    }
    box.addEventListener('click', toggle);
    box.addEventListener('keydown', function(e) {
      if (e.key === ' ' || e.key === 'Enter') { e.preventDefault(); toggle(); }
    });
  });
})();
</script>
